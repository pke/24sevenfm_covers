project(24sevenfm_covers_linux VERSION 1.0.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build this target on Linux (including WSL2/WSLg)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(COVERFETCH_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
option(COVERFETCH_BUILD_TESTS "Build portable and Linux tests" ON)
option(SSC_ENABLE_DEBUG_OVERLAY "Core diagnostics" OFF)
enable_testing()
add_subdirectory(../../lib core)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Test)
qt_add_resources(resources resources.qrc)
add_library(linux_adapter STATIC stage.cpp viewer.cpp settings.cpp ${resources})
target_link_libraries(linux_adapter PUBLIC coverfetch Qt6::Widgets)
target_compile_options(linux_adapter PRIVATE -Wall -Wextra -Wpedantic)
add_executable(24sevenfm_covers main.cpp)
target_link_libraries(24sevenfm_covers PRIVATE linux_adapter)
add_executable(linux_icon make_icon.cpp)
target_link_libraries(linux_icon PRIVATE Qt6::Gui)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/app-icon.png"
    COMMAND linux_icon "${CMAKE_CURRENT_SOURCE_DIR}/../24sevenfm_covers.ico" "${CMAKE_CURRENT_BINARY_DIR}/app-icon.png"
    DEPENDS linux_icon ../24sevenfm_covers.ico VERBATIM)
add_custom_target(linux_desktop_icon ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/app-icon.png")
if(COVERFETCH_BUILD_TESTS)
    add_executable(linux_adapter_tests tests.cpp)
    target_link_libraries(linux_adapter_tests PRIVATE linux_adapter Qt6::Test)
    add_test(NAME "Linux: adapter integration" COMMAND linux_adapter_tests)
    set_tests_properties("Linux: adapter integration" PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;SSC_TEST_MODE=1")
    add_executable(linux_transport_tests transport_tests.cpp)
    target_link_libraries(linux_transport_tests PRIVATE coverfetch)
    add_test(NAME "Linux: bounded HTTP transport" COMMAND linux_transport_tests)
endif()
install(TARGETS 24sevenfm_covers RUNTIME DESTINATION bin)
install(FILES app.dudesoft.24sevenfm.covers.desktop DESTINATION share/applications)
install(FILES README.md ../../LICENSE ../../THIRD_PARTY_NOTICES.md DESTINATION share/doc/24sevenfm-covers)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/app-icon.png" DESTINATION share/icons/hicolor/64x64/apps RENAME app.dudesoft.24sevenfm.covers.png)
set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_FILE_NAME "24sevenfm-covers-${PROJECT_VERSION}-linux-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)

# Optional production companions share the identical viewer and presentation
# adapter. The raster entry point supervises the chosen GPU process so even a
# native driver initialization crash can recover to a usable settings window.
option(SSC_GPU_RENDERERS "Include selectable Linux GPU renderers" ON)
if(SSC_GPU_RENDERERS)
    find_package(Qt6 REQUIRED COMPONENTS OpenGLWidgets)
    set(production_renderers opengl)
    if(Qt6_VERSION VERSION_GREATER_EQUAL 6.7)
        find_package(Qt6 REQUIRED COMPONENTS ShaderTools)
        list(APPEND production_renderers rhi)
    endif()
    foreach(renderer IN LISTS production_renderers)
        if(renderer STREQUAL opengl)
            set(renderer_id 1)
        else()
            set(renderer_id 2)
        endif()
        add_library(linux_adapter_${renderer} STATIC stage.cpp stage_gpu.cpp viewer.cpp settings.cpp ${resources})
        target_compile_definitions(linux_adapter_${renderer} PUBLIC SSC_RENDERER=${renderer_id})
        target_compile_options(linux_adapter_${renderer} PRIVATE -Wall -Wextra -Wpedantic)
        target_link_libraries(linux_adapter_${renderer} PUBLIC coverfetch Qt6::Widgets Qt6::OpenGLWidgets)
        add_executable(24sevenfm_covers_${renderer} main.cpp)
        target_link_libraries(24sevenfm_covers_${renderer} PRIVATE linux_adapter_${renderer})
        if(renderer STREQUAL rhi)
            target_link_libraries(linux_adapter_${renderer} PUBLIC Qt6::GuiPrivate)
            qt_add_shaders(24sevenfm_covers_rhi production_stage_shaders PREFIX "/shaders" BASE "shaders"
                FILES shaders/stage.vert shaders/stage.frag)
        endif()
        install(TARGETS 24sevenfm_covers_${renderer} RUNTIME DESTINATION bin)
    endforeach()
endif()

# Build all candidates with exactly the same Qt, controller and drawing code.
option(SSC_RENDER_BENCHMARKS "Build Linux raster/OpenGL/RHI comparison tools (Qt 6.7+)" OFF)
if(SSC_RENDER_BENCHMARKS)
    find_package(Qt6 6.7 REQUIRED COMPONENTS OpenGLWidgets ShaderTools)
    foreach(renderer IN ITEMS raster opengl rhi)
        if(renderer STREQUAL raster)
            set(renderer_id 0)
        elseif(renderer STREQUAL opengl)
            set(renderer_id 1)
        else()
            set(renderer_id 2)
        endif()
        add_executable(render_bench_${renderer} render_benchmark.cpp stage.cpp stage_gpu.cpp ${resources})
        target_compile_definitions(render_bench_${renderer} PRIVATE SSC_RENDERER=${renderer_id})
        target_compile_options(render_bench_${renderer} PRIVATE -Wall -Wextra -Wpedantic)
        target_link_libraries(render_bench_${renderer} PRIVATE coverfetch Qt6::Widgets Qt6::OpenGLWidgets)
        if(renderer STREQUAL rhi)
            target_link_libraries(render_bench_${renderer} PRIVATE Qt6::GuiPrivate)
            qt_add_shaders(render_bench_rhi stage_shaders PREFIX "/shaders" BASE "shaders"
                FILES shaders/stage.vert shaders/stage.frag)
        endif()
    endforeach()
endif()
